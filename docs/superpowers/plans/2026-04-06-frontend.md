# Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a single-page frontend served by the Go backend that displays PID controller parameters inline with an embedded Grafana dashboard.

**Architecture:** A single `web/index.html` embedded into the Go binary via a thin `web/web.go` package. The httpserver registers `GET /` to serve it. All UI logic is vanilla JS inside the HTML file — no build step, no dependencies.

**Tech Stack:** Go `//go:embed`, vanilla HTML/CSS/JS, Gin

import (
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/adrianozp/gaardrail/internal/httpserver"
)

func TestGetIndex(t *testing.T) {
	router := httpserver.New()

	w := httptest.NewRecorder()
	req, _ := http.NewRequest(http.MethodGet, "/", nil)
	router.ServeHTTP(w, req)

	if w.Code != http.StatusOK {
		t.Errorf("expected status 200, got %d", w.Code)
	}

	ct := w.Header().Get("Content-Type")
	if !strings.HasPrefix(ct, "text/html") {
		t.Errorf("expected Content-Type text/html, got %q", ct)
	}

	if !strings.Contains(w.Body.String(), "<html") {
		t.Error("expected response body to contain <html")
	}
}
```

package httpserver

import (
	"net/http"
	"time"

	"github.com/adrianozp/gaardrail/web"
	"github.com/gin-gonic/gin"
	"github.com/rs/zerolog/log"
)

func New() *gin.Engine {
	router := gin.New()
	router.Use(metricsMiddleware())

	router.GET("/", func(c *gin.Context) {
		c.Data(http.StatusOK, "text/html; charset=utf-8", web.IndexHTML)
	})

	router.GET("/ping", func(c *gin.Context) {
		c.Status(http.StatusOK)
	})

	return router
}

func metricsMiddleware() gin.HandlerFunc {
	return func(c *gin.Context) {
		start := time.Now()
		c.Next()
		log.Debug().
			Str("method", c.Request.Method).
			Str("path", c.Request.URL.Path).
			Dur("duration_ms", time.Since(start)).
			Msg("request")
	}
}
```

    const FIELDS = ['kp', 'ki', 'kd', 'setpoint', 'min', 'max', 'i_clamp'];
    let current = {};

    async function loadParams() {
      try {
        const res = await fetch('/pid');
        if (!res.ok) throw new Error(res.statusText);
        current = await res.json();
        renderView();
      } catch (_) {
        renderError();
      }
    }

    function renderView() {
      document.getElementById('params').innerHTML = FIELDS.map(f => `
        <div class="param">
          <label>${f}</label>
          <span class="value">${current[f] !== undefined ? current[f] : '—'}</span>
        </div>
      `).join('');
      show('btn-edit');
      hide('btn-cancel');
      hide('btn-save');
      setBarClass('');
    }

    function renderError() {
      document.getElementById('params').innerHTML = `
        <span style="color:#da3633;font-size:12px">Erro ao carregar parâmetros</span>
        <button onclick="loadParams()"
          style="background:transparent;border:1px solid #30363d;color:#8b949e;
                 padding:2px 8px;font-size:11px;border-radius:3px;cursor:pointer;margin-left:8px">
          ↺ Tentar novamente
        </button>
      `;
      hide('btn-edit');
      hide('btn-cancel');
      hide('btn-save');
    }

    function enterEdit() {
      document.getElementById('params').innerHTML = FIELDS.map(f => `
        <div class="param">
          <label>${f}</label>
          <input type="number" step="any" id="input-${f}" value="${current[f] !== undefined ? current[f] : 0}">
        </div>
      `).join('');
      hide('btn-edit');
      show('btn-cancel');
      show('btn-save');
      setBarClass('editing');
    }

    function cancelEdit() {
      renderView();
    }

    async function saveParams() {
      const body = {};
      for (const f of FIELDS) {
        const input = document.getElementById(`input-${f}`);
        const val = parseFloat(input.value);
        if (val !== current[f]) body[f] = val;
      }

      if (Object.keys(body).length === 0) {
        renderView();
        return;
      }

      try {
        const res = await fetch('/pid', {
          method: 'PATCH',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(body),
        });

        if (!res.ok) {
          const data = await res.json().catch(() => ({}));
          showFeedback('error', data.error || `Erro ${res.status}`);
          return;
        }

        for (const [k, v] of Object.entries(body)) current[k] = v;
        showFeedback('success', null);
        setTimeout(renderView, 2000);
      } catch (_) {
        showFeedback('error', 'Erro de rede');
      }
    }

    function showFeedback(type, msg) {
      setBarClass(type);
      const el = document.getElementById('error-msg');
      if (msg) {
        el.textContent = msg;
        el.style.display = '';
        setTimeout(() => { el.style.display = 'none'; }, 3000);
      }
    }

    function setBarClass(cls) {
      document.getElementById('bar').className = cls;
    }

    function show(id) { document.getElementById(id).style.display = ''; }
    function hide(id) { document.getElementById(id).style.display = 'none'; }

    loadParams();
  </script>
</body>
